QCMS是一款小型的网站管理系统。拥有多种结构类型，包括：ASP+ACCESS、ASP+SQL、PHP+MYSQL

private function _move_file($file, $ext){
        $url = $this->_dir.$this->_name.'.'.$ext;
        if(!is_dir($this->_dir)){
            mkdir($this->_dir, 0777, true);
        }
        if (!move_uploaded_file($file, $url)){
