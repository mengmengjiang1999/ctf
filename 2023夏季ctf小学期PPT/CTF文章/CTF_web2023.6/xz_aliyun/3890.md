def newaes(key):
    aes_key = 0
    while key > 0:
        aes_key ^= key & (2 ** 256 - 1)
        key >>= 256
    try:
        aes_key = ('%x' % aes_key).strip('L').decode('hex')
    except:
        aes_key = ('0%x' % aes_key).strip('L').decode('hex')
    aes = AES.new(aes_key, AES.MODE_CBC, '0' * 16)
    return aes

def dec(key,encmsg):
    msg = newaes(key).decrypt(encmsg.strip().decode('base64'))
