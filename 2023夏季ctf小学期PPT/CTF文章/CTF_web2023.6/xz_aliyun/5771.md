xssflag.py:

#/usr/bin/env python
# _*_ coding:utf-8 _*_
# blog : evilwing.me
# __Author__ : wing
from flask import Flask, request

xss = Flask(__name__)


@xss.route('/')
def index():
    flag = request.args
    for i,j in flag.items():
        print('Flag is:' + j)
    return str()


if __name__ == "__main__":
    xss.run(host="0.0.0.0")
