#include<string>
using namespace std;
class Student
{
private:
	int number;
	string name;
	int chinese;
	int maths;
	int english;
	float av;
public:
	void member(int n,string na,int c,int m,int e);

		av=(chinese+maths+english)/3;

	 void print()
	 {

		 cout<<"学号是:"<<number<<endl;

		 cout<<"数学为:"<<maths<<endl;
		 cout<<"英语为:"<<english<<endl;
		 cout<<"平均分为:"<<av<<endl;
	 }
};
void Student::member(int n,string na,int c,int m,int e)
	{
    number=n;
	name=na;
	chinese=c;
	maths=m;
	english=e;
	}
