#同时使用用户代理池与ip代理池
import urllib.request
import re
import random #实现随机选择的模块

uapools=[					#构建用户代理池
        用户代理1
        用户代理2
        用户代理3
]
def ip():					#创建ip代理函数
    print("这一次调用了接口")
    thisip=urllib.request.urlopen("提供代理ip网站url").read().encode("utf-8","ignore")
    ippools=[]
    for item in thisip:
        ippools.append(item.decode("utf-8","ignore"))
    return ippools			#通过ip代理网站获取ip
def ip (ippools,time,uapools):
    rst = random.choice(uapools)#随机获取用户代理
    print(rst)
    headers = ("User-Agent", rst)
    thisip=ippools[time]	#根据时间改变改变获取的ip
    print("当前用户ip为："+ippools[time])
    proxy=urllib.request.ProxyHandler({"http":thisip})		#创建头文件信息，将代理信息存入
    opener=urllib.request.build_opener(proxy,urllib.request.HTTPHandler)
    opener.addheaders = [headers]
    urllib.request.install_opener(opener)

x=0
for i in range(0,3):		#执行爬取命令
    try:
        if (x % 10 == 0):
            time = x % 10
            ippools = ip()
            ip(ippools, time, uapools)
        else:
            time = x % 10
            ip(ippools, time ,uapools)
        url = "http://www.baidu.com"
        data1 = urllib.request.urlopen(url).read()
        data = data1.decode("utf-8", "ignore")
        print(len(data))
        fh = open("保存目录", "w")
        fh.write(data1)
        fh.colse()
        x += 1
    except Exception as err:
        print(err)
        x += 1


2.Scrapy框架

爬虫可以根据复杂程度分为:爬虫项目与爬虫文件，虽然可以利用urllib模块进行爬虫的实现，但是对于大型项目来说效率不高并且稳定性存在问题

Scrapy是一个Python爬虫框架，使用Scrapy可以提高开发效率，非常适合做中大型爬虫项目。

所以urllib适合做爬虫文件，但是scrapy适合做爬虫项目。

3.XPath表达式(一般用于对xml文件的提取)

(与正则表达式相同，XPath表达式同样也是进行匹配的表达式)

  * 1.XPath表达式效率相对更高一些

  * 2.正则表达式更加强大一些

  * 3.一般有限选择XPath表达式，但是XPath解决不了的问题利用正则表达式解决

1）XPath基础语法

首先介绍XPath中的基础语法

/ ：逐层提取

​ 当我们在一个html文件中提取信息，由于HTML文件中利用标签来定义信息，所以利用/符号来进行操作

例如:

​ 需要提取标题：/html/head/title 此时会将title内容以及标签提取出来

​ 但是如果我们只需要提取title中的内容应该怎么做呢?可以利用text()标签来进行操作**

text() ：提取标签下面的文本

​ 此时如果只要提取标题下面的文本就可以这样构造：

​ /html/head/title/text()

//标签名## ：提取所有名为## 的标签

例如此图:

当我们需要调用meat标签时，可以//meta,来进行提取所有的meta标签

但是如果我没只想要第二个meta标签，此时应该怎么做呢

我们可以构造**//标签名[@属性=‘属性值’]**来进行提取

例如此时可以构造：//meta[@http-equiv='Content-Security-Policy']，因为此属性的值是唯一的，所以我们这样构造来进行提取

@属性名 ：代表提取某个属性的值

例如：

div中的属性：< div class="tools" > 标签中的内容： //div[@class="tools"]，就如上面我们进行构造的一样，可以进行针对性的提取

例如此时我们需要爬取一个li 中的内容,例如首页,但是搜索class，发现页面中有重复的内容，于是此时我们可以查看前一层标签ul，发现其class是唯一的,于是此时构造**//ul[@class='item nav-list clearfix']/li/text()**此时就可以爬取出li标签的内容了，所以还是需要有思路，爬虫什么的都是很简单的

5.利用scrapy模拟登录爬虫

利用scrapy框架代替人登录网站，之后可以爬取深层次页面。

这是自己定义的爬虫文件的内容

import scrapy
from scrapy.http import Resquest,FormRequest


class MonidengluSpider(scrapy.Spider):
    name = 'monidenglu'
    allowed_domains = ['xxx.com']
    start_urls = ['http://xxx.com/']
    header = {"User-Agent":"Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:95.0) Gecko/20100101 Firefox/95.0"}
    #编写start_requests()方法，第一次会默认调取该方法中的请求,如果不进行定义则首次访问地址为start_urls中的地址
    def start_requests(self):
        # 首先爬一次登录页，然后回调函数parse()，cookiejar参数设置为一的目的是让爬虫可以利用计算机用户的cookie使其可以防止被拦截
        # 回调函数是指爬取的数据之后转交给parse函数进行处理
        return [Resquest("http://edu.xxx.com/index_user_login",meta={"cookiejar":1},callback=self.parse)]

    def parse(self, response):
        #设置要传递的post信息，没有验证码字段
        data={
            "number":"lidalu",
            "passwd":"lidalu123",
        }
        print("登录中......")
        #通过FormRequest.from_response()进行登陆,利用之前设置的参数信息进行请求
        return [FormRequest.from_response(response,
                                          #设置cookie信息
                                          meta={"cookiejar":response.meta["cookiejar"]},
                                          #设置headers信息模拟成浏览器
                                          headers=self.header,
                                          #设置post表单中的数据
                                          format=data,
                                          #设置回调函数，此时回调函数为next()
                                          callback=self.next,
                                        )]
        #此时再次定义回调函数
        def next(self,response):
            #由于response只是一个响应信息，而不是字符串，所以此时无法利用正则表达式去匹配，之后将响应转交给一个变量之后才可以进行正则匹配
            #a=response.body
            #此时response指的是登陆成功界面
            print(response.xpath("/html/head/title/text()").extract())
            #设置完成之后为了保证登录成功，对其个人主页进行访问
            yield Resquest("http://edu.xxx.com/index_user_index",callback=self.next2,meta={"cookiejar":True})
        def next2(self,response):
            #此时response指的是个人主页页面
            print(response.xpath("/html/head/title/text()").extarct())
            #最后输出的是用户中心的title

1）半自动输入：

当我们登录某些网站时，有时多次登录之后会提示我们输入验证码，此时我们利用爬虫进行验证码自动输入

与之前自动登录无异，但是需要在登录之前添加一个核验是否有验证码需要输入的操作

一般图文验证码以图片的形式出现，如图:

此时我们从源码中获取验证码的对应地址进行爬取存入本地，之后从本地进行识别输出

如下，其中parse函数中首先对于验证码进行了验证与下载

import urllib.request

import scrapy
from scrapy.http import Resquest,FormRequest

class D1Spider(scrapy.Spider):
    name = 'd1'
    allowed_domains = ['douban.com']
    start_urls = ['http://douban.com/']
    header = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:95.0) Gecko/20100101 Firefox/95.0"}
    def start_requests(self):
        # 首先爬一次登录页，然后回调函数parse()，cookiejar参数设置为一的目的是让爬虫可以利用计算机用户的cookie使其可以防止被拦截
        # 回调函数是指爬取的数据之后转交给parse函数进行处理
        return [Resquest("https://accounts.douban.com/passport/login",meta={"cookiejar":1},callback=self.parse)]

    def parse(self, response):
        #判断是否有验证码
        captcha=response.xpath("//img[id='captcha_image']/@src").extract()
        if len(captcha)>0:
            print("此时有验证码")
            #将验证码保存到本地
            localpath="本地路径"
            urllib.request.urlretrieve(captcha[0],filename=localpath)
            captcha_value=input"请到(本地路径)查看验证码图片"
        # 设置要传递的post信息，没有验证码字段
        #有验证码字段通过查看返回的包进行判断添加其他关于验证码参数
        data = {
            "redir":"https://accounts.douban.com/passport/login",
            "from_number": "gingingg",
            "form_password": "lixiaojie001124",
        }
        print("登录中......")
        # 通过FormRequest.from_response()进行登陆,利用之前设置的参数信息进行请求
        return [FormRequest.from_response(response,
                                          # 设置cookie信息
                                          meta={"cookiejar": response.meta["cookiejar"]},
                                          # 设置headers信息模拟成浏览器
                                          headers=self.header,
                                          # 设置post表单中的数据
                                          format=data,
                                          # 设置回调函数，此时回调函数为next()
                                          callback=self.next,
                                          )]
        def next(self,response):
            title=response.xpath("/html/head/title/text()").extract()
            print(title)
