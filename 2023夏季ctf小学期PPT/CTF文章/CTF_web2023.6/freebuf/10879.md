扫描器开发从0到1--服务爆破
2022-05-30 16:31:26

服务爆破本质上也很简单，当我们知道目标服务是开的，例如3306，我们知道他的端口是开放的，那我们我们就可以通过tcp连接向他发送认证数据包，认证成功之后就会建立一个服务类型的长连接。这里我们可以用已经有的第三方库来连接，也可以使用net库建立连接自己构造数据包来进行认证。

使用第三方库方便简单，一个函数基本上就可以搞定，并且功能强大像数据库这种服务还可以执行命令。但是缺点就是引入的数据库驱动体积会比较大。

衡量了一下还是选择的第三方库，毕竟简单方便功能强大，如果以后有体积需求再说。

func ssh_connect_userpass(addr,user,pass string) (*ssh.Client,error) {
  //创建一个ssh客户端配置
    client_config:=&ssh.ClientConfig{User: user,     //用户名
                                   Auth: []ssh.AuthMethod{ssh.Password(pass)},   //认证方式
                                   HostKeyCallback: ssh.InsecureIgnoreHostKey(),
                                   Timeout: Timeout}   
    conn,err:=Getconn(addr)   //给一个底层的tcp连接
    if err!=nil{
        return nil,err
    }
    c,ch,re,err:=ssh.NewClientConn(conn,addr,client_config) //在给定的底层连接上建立ssh连接
    if err !=nil{
        return nil,err
    }
    return ssh.NewClient(c,ch,re), nil     //进行身份认证
}

爆破部分

爆破的就是我们针对这个ip的这个端口不断的发送身份验证数据包，根据返回包来判断是否认证成功，也就是多线程调用我上面列出的函数。

同样使用端口扫描的思路，只不过多了一步从本地文件中读取用户名或者密码文件

 1. 获取目标ip和端口
 2. 读取用户名密码文件
 3. 将用户名密码放入任务列表中
 4. 开启多线程运行认证函数
 5. 爆破成功之后所有线程停止
 6. 返回结果

需要注意的问题：

  * 读取密码文件的时候，不要想密码文件直接转换成列表放到内存里面，要通过端口扫描的那种处理方式，放到队列或者信道中
  * 开启多线程爆破的时候如果爆破成功了的话，如何停掉其他所有的线程，并停止这次任务，还需要停止读取密码文件的线程

爆破结构体

type Burp struct {
    password_ch chan string    //存放密码信道
    username_list []string       //存放用户名
    username string           //接收的用户名
    password string           //接收的密码
    userdict string           //用户名字典名
    passdict string           //密码字典名
    aliveaddr string          //目标地址
    tasklist chan *burp_info      //任务列表，存放着用户名密码和目标地址
    service Service           //身份认证函数
    wg sync.WaitGroup        //控制线程
    burpthread int           //爆破的线程数
    stop chan int8           //要来暂停扫描
    burpresult string        //用来返回结果
}


构造函数：需要传入用户名和密码目标ip，认证函数，爆破线程

func NewBurp(pass,user,userdict,passdict string,aliveaddr string,service Service,burpthread int) (*Burp) {
    return &Burp{password_ch: make(chan string,Thread*2),
        username_list: []string{},
        userdict: userdict,
        passdict: passdict,
        tasklist: make(chan *burp_info,Thread*2),
        service: service,
        aliveaddr: aliveaddr,
        password:pass,
        username: user,
        burpthread: burpthread,
        stop:make(chan int8),
    }
}
type Service func(user string,pass string,addr string)(error,bool,string)

根据用户名密码生成任务列表

func (b *Burp) Gettasklist()  {
    defer b.wg.Done()
    for pass:=range b.password_ch{
        for _,user:=range b.username_list{
            if cancelled(b.stop) {    //判断是否爆破成功
                break
            }
            b.tasklist<-&burp_info{user,pass,b.aliveaddr}
        }
    }
    close(b.tasklist)
}


进行多线程爆破

func (b *Burp) Check()  {
    defer b.wg.Done()
    for task:=range b.tasklist{       //判断是否爆破成功
        if cancelled(b.stop) {
            break
        }
        if Verbose{
            fmt.Println(Yellow(fmt.Sprintf("Test:%v %v %v",task.addr,task.username,task.password)))
        }
        err,success,servername:=b.service(task.username,task.password,task.addr)  //利用传进来的认证函数进行爆破
        if err==nil&&success{
            if cancelled(b.stop){    //这步判断是为了防止为授权这种多次爆破成功然后多次输出情况
                break
            }else {
                Output(fmt.Sprintf("\r[+]%v burp success:%v %v %v\n",servername,task.addr,task.username,task.password),LightGreen)
                if cancelled(b.stop) {   //同上
                    break
                }
                b.burpresult=fmt.Sprintf("Username:%v\tPassword:%v",task.username,task.password)
                close(b.stop)
                break
            }
        }
    }
}


run函数

func (b *Burp) Run() string {
    switch  {
    case b.username!=""&&b.userdict=="":
        if strings.Contains(b.username,","){
            userlist:=strings.Split(b.username,",")
            for _,user:=range userlist{
                b.username_list=append(b.username_list,user)
            }
        }else {
            b.username_list=append(b.username_list,b.username)
        }
    case b.userdict!="":
        b.Getuser()
    default:
        b.username_list=[]string{""}
    }
    switch  {
    case b.password==""&&b.passdict!="":
        b.wg.Add(1)
        go b.Getpass()
    case b.password!=""&&b.passdict=="":
        if strings.Contains(b.password,","){
            passlist:=strings.Split(b.password,",")
            for _,pass:=range passlist{
                b.password_ch<-pass
            }
            close(b.password_ch)
        }else {
            b.password_ch<-b.password
            close(b.password_ch)
        }
    default:
        b.password_ch=make(chan string,len(pass_dict))
        for _,i:=range pass_dict{
            b.password_ch<-i
        }
        close(b.password_ch)
    }             //上面都是判断获取用户名和密码的细节问题，会将用户名 密码写入信道和列表中
    b.wg.Add(1)
    go b.Gettasklist()          //开启线程获取任务列表
    for i:=0;i<b.burpthread;i++{
        b.wg.Add(1)       
        go b.Check()            //多线程开始爆破
    }
    b.wg.Wait()
    return b.burpresult   //返回结果
}

func Connectmysql(ip string, port int) (string, int, error,[]string) {
    conn, err := Getconn(fmt.Sprintf("%v:%v",ip,port))    //获取tcp连接
    if conn != nil {
        defer conn.Close()
        fmt.Printf(White(fmt.Sprintf("\rFind port %v:%v\r\n", ip, port)))
        fmt.Println(Yellow("Start burp mysql : ",ip))
        _,f,_:=mysql_auth("zxczxc","zxczxc",ip)      //判断为授权
        if f{
            Output(fmt.Sprintf("%v burp success:%v No authentication\n","mysql",ip),LightGreen)
            return ip,port,nil,[]string{"No authentication"}
        }
        startburp:=NewBurp(Password,Username,Userdict,Passdict,ip,mysql_auth,burpthread) //这两行会在当前这个扫描的线程中开启多线程爆破
        startburp.Run()
    }
    return ip, port, err,nil
}
