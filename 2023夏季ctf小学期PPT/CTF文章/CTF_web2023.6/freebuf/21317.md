序列化和反序列化

序列化后的格式

  类型                                                                    结构                                                                
String                                                             s:size:value;                                                          
Integer                                                              i:value;                                                             
Boolean                                                           b:value;(保存1或0)                                                         
 Null                                                                   N;                                                                
 Array                                    a:size:{key definition;value definition;(repeated per element)}                                 
Object   O:strlen(object name):object name:object size:{s:strlen(property name):property name:property definition;(repeated per property)}

//序列化函数serialize()   反序列化函数unserialize()
//注意加引号和不加引号，序列化后的区别 ，加引号后会被当成字符串
//不加引号
<?php
$key=123;
echo serialize($key);
?>
//执行结果   i:123;


//加引号
    <?php
$key='123';
echo serialize($key);
?>
   //执行结果   s:3:"123";

我们看到变量KEY在后面才被赋值，前面是没有赋值的，也就是说 key的值为空 ，那么写题思路就是burp抓包 cookie变量设为字符串类型的空（因为"$KEY"），然后序列化 后赋值给 cookie变量

<?php
$_COOKIE="";
echo serialize($_COOKIE);
?>

//执行结果： s:0:"";

class FileHandler {

    protected $op;
    protected $filename;
    protected $content;

    function __construct() {
        $op = "1";
        $filename = "/tmp/tmpfile";
        $content = "Hello World!";
        $this->process();
    }

    public function process() {
        if($this->op == "1") {
            $this->write();
        } 
        else if($this->op == "2") {    //因为op=2  这里弱比较  所以判断结果为true
            $res = $this->read();      //调用read函数
            $this->output($res);       //将变量res的值输出
        }
        else {
            $this->output("Bad Hacker!");
        }
    }

    private function write() {
        if(isset($this->filename) && isset($this->content)) {
            if(strlen((string)$this->content) > 100) {
                $this->output("Too long!");
                die();
            }
            $res = file_put_contents($this->filename, $this->content);
            if($res) $this->output("Successful!");
            else $this->output("Failed!");
        } else {
            $this->output("Failed!");
        }
    }

    private function read() {
        $res = "";
        if(isset($this->filename)) {
            $res = file_get_contents($this->filename);/*看到file_get_contents()这个函数应该可以明白需要用php伪协议读*/                        }                       /*取文件内容，也就是说使用php伪协议来读取flag.php文件的内容  */
        return $res;
    }

    private function output($s) {
        echo "[Result]: <br>
";
        echo $s;
    }

    function __destruct() {
        if($this->op === "2")   //不能让这个条件成立  输入op=2 导致强比较 结果为false，所以这里op不会被覆盖成1
            $this->op = "1";
        $this->content = "";  //content置为空
        $this->process();     //调用process（）函数
    }

}
