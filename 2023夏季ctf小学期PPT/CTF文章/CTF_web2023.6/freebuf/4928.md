在说明这个字符串数组的作用之前不得不说一个前情提要：在老版本的nmap中，是支持以类似于以下结果的IP作为目标地址的 ：138.[1-7,16,91-95,200-].12.1（经尝试目前版本不支持，可能由于不稳定的原因）

如果输入的是上述类型的IP，那么对于IP中4个部分每一个部分中都有可能出现类似于[1-7,16,91-95,200-]这样范围的指定，所以势必就要对每部分进行分别解析，最终再汇总所有涉及到的IP地址，那么如何解析这串复杂的 "IP" 呢，就从上述定义的字符串数组入手，将IP以.为分割符，分成4个部分，并分别解析其每一段的特殊范围即可

但这种特殊的只是一种情况，也存在正常的IP所以接下来势必会有条件判断来区分这两种情况，后面会依次说明，所以上述变量的定义主要完成了这样的功能：

  * char *hostexp = strdup(h);将传进来的 IP/IP范围/域名字符串h的内容在堆中重新开辟一片空间，并将这个指针赋给hostexp

  * addy[0] = r = hostexp;将这个重新申请的字符串指针赋给r与addy[0]

    其中r就是当输入的IP应用了上述所受的范围时，用于对IP进行拆分时用到的备份指针

此处对于当前字符串是域名还是IP用了一个很简单的判断方式：

利用一个下边遍历字符串数组中的每一个字节，若存在 大写/小写字母 则认为是一个域名，否则则是一个IP地址

为了区分如下两种情况，所以直接使用inet_aton(target_net, &(targets->start))

上面说了，对于此处传入的字符串若为IP的话有两类，一类是正常的IP不带范围，另一类是附带格式类似于[]的范围，而代码先要处理的是正常的IP

// 将刚刚放入的IP转为主机字节序

// 计算加上掩码后的范围

        targets->currentaddr = targets->start;

        if (targets->start.s_addr <= targets->end.s_addr)
        {
            free(hostexp);
            return 1;
        }
        fprintf(stderr, "Host specification invalid");
        free(hostexp);
        return 0;
    }
    else
    {
        i=0;
        targets->maskformat = 0;
        while(*++r)
        {
            if (*r == '.' && ++i < 4)
            {
                *r = '\\0';
                addy[i] = r + 1;
            }
            else if (*r == '[')
            {
                *r = '\\0';
                addy[i]++;
            }
            else if (*r == ']')
            {
                *r = '\\0';
            }
            else if (*r != '*' && *r != ',' && *r != '-' && !isdigit((int)*r))
            {
                fatal("Invalid character in  host specification.");
            }
        }
        if (i != 3) fatal("Target host specification is illegal.");
        for(i=0; i < 4; i++)
        {
            j=0;
            while((s = strchr(addy[i],',')))
            {
                *s = '\\0';
                if (*addy[i] == '*')
                {
                    start = 0; end = 255;
                }
                else if (*addy[i] == '-')
                {
                    start = 0;
                    if (!addy[i] + 1)
                    {
                        end = 255;
                    }
                    else
                    {
                        end = atoi(addy[i]+ 1);
                    }
                }
                else
                {
                    start = end = atoi(addy[i]);
                    if ((r = strchr(addy[i],'-')) && *(r+1) )
                    {
                        end = atoi(r + 1);
                    }
                    else if (r && !*(r+1))
                    {
                        end = 255;
                    }
                }
                if (o.debugging)
                    printf("The first host is %d, and the last one is %d\\n", start, end);
                if (start < 0 || start > end)
                {
                    fatal("Your host specifications are illegal!");
                }
                for(k=start; k <= end; k++)
                {
                    targets->addresses[i][j++] = k;
                    addy[i] = s + 1;
                }
            }
            if (*addy[i] == '*') { start = 0; end = 255; }
            else if (*addy[i] == '-') {
            start = 0;
            if (!addy[i] + 1) end = 255;
            else end = atoi(addy[i]+ 1);
            }
            else
            {
                start = end = atoi(addy[i]);
                if ((r =  strchr(addy[i],'-')) && *(r+1) ) end = atoi(r+1);
                else if (r && !*(r+1)) end = 255;
            }
            if (o.debugging)
            printf("The first host is %d, and the last one is %d\\n", start, end);
            if (start < 0 || start > end) fatal("Your host specifications are illegal!");
            if (j + (end - start) > 255) fatal("Your host specifications are illegal!");
            for(k=start; k <= end; k++)
            targets->addresses[i][j++] = k;
            targets->last[i] = j - 1;
    }
    }
    bzero((char *)targets->current, 4);
    free(hostexp);
    return 1;
}
