{

	cout<<"hour:"<<hour<<" minute:"<<minute<<" second:"<<second<<endl;

	static void ShowTotalTime();
	~Time(){}

	int hour;
	int minute;
	int second;
	
};
void Time::ShowTotalTime()

	totalTime=hour*60*60+minute*60+second;
	}

3.类中静态成员的使用方法

#include<iostream>
using namespace std;
class A
{
public:
	A(int i)
	{
		a = i;
	}
	int geta()
	{
		return a;
	}
	static int getb()
	{
		return b;
	}
	static void setb()
	{
		b = b + 1;
	}
private:
	int a;
	static int b;
};
int A::b = 10;
int main()
{
	A a1(1),a2(2);
	cout << a1.geta() << "," << a1.getb() << endl;
	cout << a2.geta() << "," << a2.getb() << endl;
	a1.setb();
	A::setb();
	cout << a1.geta() << "," << a1.getb() << endl;
	cout << a2.geta() << "," << a2.getb() << endl;
	return 0;
}

#include <iostream>

{
public:

Boat(int a)
{weight=a;}
};
class Car
{
public:
	int weight;
	Car(int c)
	{weight=c;}
	int getTotalWeight(Boat b);
};
int Car::getTotalWeight(Boat b)
{
int T;
T=b.weight+weight;
return T;
}

{
int TT;
